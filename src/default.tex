\documentclass[
  paper=a4,
  fontsize=11pt,
  twoside=true,
  headings=normal,
  DIV=12
]{scrbook}

% ------------------------------------------------------------
% Engine check: Pandoc must use LuaLaTeX
% ------------------------------------------------------------
\usepackage{ifluatex}
\ifluatex\else
  \errmessage{This template requires LuaLaTeX. Use --pdf-engine=lualatex}
\fi

% ------------------------------------------------------------
% Font setup (TeX Gyre Pagella = Palatino clone)
% ------------------------------------------------------------
\usepackage{fontspec}
\defaultfontfeatures{Ligatures=TeX, Scale=MatchLowercase}

\setmainfont{TeX Gyre Pagella}
\setsansfont{TeX Gyre Heros}
\setmonofont{TeX Gyre Cursor}

% ********************************************************************
% Geometry
% ********************************************************************
\usepackage{geometry}
\geometry{
  paper=a4paper,
  margin=25mm,
  headheight=14pt
}

% --------------------------------------------------------------------
% centertextmargins
% --------------------------------------------------------------------
% Environment for setting margins evenly on a single page so that the
% left and right margins are equal, without affecting the top and
% bottom margins, and maintaining the text area width.
% --------------------------------------------------------------------
\newenvironment{centertextmargins}
    {\newgeometry{left=0.2\paperwidth, right=0.2\paperwidth, top=2.5cm, bottom=3.5cm}}
    {\restoregeometry}

% ------------------------------------------------------------
% Microtypography
% ------------------------------------------------------------
\usepackage{microtype}

% ------------------------------------------------------------
% Packages to support Pandoc
% ------------------------------------------------------------

% Pandoc list tightening
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Hyperlinks and anchors (required for Pandoc)
\usepackage[hidelinks]{hyperref}

% Formatting for URLs
\urlstyle{same}

% ********************************************************************
% Headings and heading pages
% ********************************************************************

% Heading numbering
\setcounter{secnumdepth}{2}

% Heading font
\setkomafont{disposition}{\rmfamily}

% --------------------------------------------------------------------
% Part
% --------------------------------------------------------------------
% Renew part command to handle different implementations of part command
% 
% \part{title}
% \part*{title}
% \part[toc-title]{title} % toc-title set to title if missing
% 
% https://tex.stackexchange.com/questions/4386/defining-starred-versions-of-commands-macro
% https://tex.stackexchange.com/questions/47351/can-i-redefine-a-command-to-contain-itself
% 
% See page 493 KOMA manual
% --------------------------------------------------------------------

% Typography
\setkomafont{part}{\scshape\Large}
\setkomafont{partprefix}{\Large}

% Spacing
\RedeclareSectionCommand[
  innerskip=11pt
]{part}

% TODO -- sort out spacing and kerning for MakeUppercase

% Prefix + title formatting
\renewcommand*{\partlineswithprefixformat}[3]{%
  \centering
  \parbox{.75\linewidth}{\raggedpart #2\par}%
  \par
  \parbox{.75\linewidth}{\raggedpart \color{Maroon}\MakeUppercase{#3}\par}%
}

\AddToHook{cmd/part/before}{\begin{centertextmargins}}
\AddToHook{cmd/part/after}{\end{centertextmargins}}

% Page style for part pages
\newpagestyle{partstyle}{
  \sethead{}{}{}
  \setfoot{}{}{}
}
\renewcommand*\partpagestyle{partstyle}

% ------------------------------------------------------------
% Pandoc inserts its own packages here
% ------------------------------------------------------------
$if(highlighting-macros)$
$highlighting-macros$
$endif$

$if(tables)$
\usepackage{longtable,booktabs}
$endif$

$if(graphics)$
\usepackage{graphicx}
$endif$

% ------------------------------------------------------------
% Document metadata
% ------------------------------------------------------------
\title{$title$}
\author{$author$}
\date{$date$}

% ------------------------------------------------------------
% Begin document
% ------------------------------------------------------------
\begin{document}

\frontmatter
\maketitle
\tableofcontents

\mainmatter
$body$

\end{document}
